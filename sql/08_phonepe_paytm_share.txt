WITH apps AS (
  SELECT
    LOWER(application_name) AS application_name,
    total_volume_mn,
    total_value_cr
  FROM `phonepe-paytm-analytics.phonepe_paytm.upi_apps_july_2026`
  WHERE LOWER(application_name) IN ('phone pe', 'paytm')
),

comparison AS (
  SELECT
    SUM(total_volume_mn) AS combined_volume_mn,
    SUM(total_value_cr) AS combined_value_cr
  FROM apps
)

SELECT
  a.application_name,
  a.total_volume_mn,
  ROUND(
    SAFE_DIVIDE(a.total_volume_mn, c.combined_volume_mn) * 100,
    2
  ) AS combined_volume_share_pct,
  a.total_value_cr,
  ROUND(
    SAFE_DIVIDE(a.total_value_cr, c.combined_value_cr) * 100,
    2
  ) AS combined_value_share_pct
FROM apps AS a
CROSS JOIN comparison AS c
ORDER BY a.total_volume_mn DESC;
