SELECT
  CASE
    WHEN LOWER(application_name) LIKE '%phonepe%' THEN 'PhonePe'
    WHEN LOWER(application_name) LIKE '%google pay%' THEN 'Google Pay'
    WHEN LOWER(application_name) = 'paytm' THEN 'Paytm'
    ELSE TRIM(REGEXP_REPLACE(application_name, r'\s*#\s*$', ''))
  END AS application_name,

  SUM(SAFE_CAST(total_value_cr AS FLOAT64))
  /
  NULLIF(SUM(SAFE_CAST(total_volume_mn AS FLOAT64)), 0)
  AS average_transaction_value

FROM `phonepe-paytm-analytics.phonepe_paytm.upi_apps_historical_2022_2026`

WHERE application_name IS NOT NULL

GROUP BY application_name

ORDER BY average_transaction_value DESC;
